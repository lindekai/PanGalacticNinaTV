# M4.1 The Night Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A "Night" page and a live Object module on the Apple TV that show tonight's darkness, the target's altitude curve, Sun, Moon and a compass, computed on the device from the site, the clock and the target's coordinates.

**Architecture:** A pure astronomy engine in `PanGalacticCore/Astronomy/` (Meeus algorithms, no I/O) feeds a pure `NightPlan`/`NightNow` model. The target comes from `/sequence/state` (already polled) or the tracking mount; the site from `/equipment/info` (already polled), cached per instance on the device by a `SiteStore` in `PanGalacticServices`. SwiftUI draws the page and the module with `Canvas`.

**Tech Stack:** Swift 6, SwiftUI, tvOS 26, Swift Testing, Swift package `PanGalacticCore` (targets `PanGalacticCore`, `PanGalacticServices`).

**Spec:** [`docs/superpowers/specs/2026-11-20-m4-night-design.md`](../specs/2026-11-20-m4-night-design.md)

## Global Constraints

- No new endpoint; the allowlist (ADR 0001) is unchanged. No write to PHD2, no application messages on `/v2/socket`.
- No dependency: the engine is own Swift code after Jean Meeus, *Astronomical Algorithms* (2nd ed.).
- Angles are degrees at every public API; azimuth is measured from north through east; longitude is east-positive.
- ΔT is a fixed 69 s (`JulianDay.deltaT`).
- `mount/info` site `0, 0, 0` means "no site", never a site on the equator.
- The site is never logged, exported, committed, shown in screenshots or sent anywhere; `Site`'s `description` is redacted.

```swift
import Foundation

// Degree-based trigonometry for the astronomy engine (Meeus works in degrees).
func sinDeg(_ x: Double) -> Double { sin(x * .pi / 180) }
func cosDeg(_ x: Double) -> Double { cos(x * .pi / 180) }
func tanDeg(_ x: Double) -> Double { tan(x * .pi / 180) }
func asinDeg(_ x: Double) -> Double { asin(max(-1, min(1, x))) * 180 / .pi }
func atan2Deg(_ y: Double, _ x: Double) -> Double { atan2(y, x) * 180 / .pi }

/// 0 ..< 360.
func normalizedDegrees(_ x: Double) -> Double {
    let r = x.truncatingRemainder(dividingBy: 360)
    return r < 0 ? r + 360 : r
}
```

```swift
import Foundation

/// Julian Day (Meeus ch. 7). Dynamical time is UT + ΔT with a fixed ΔT.
public enum JulianDay {
    /// TT − UT in seconds; about 69 s in the 2020s, an error below 1 s of time until 2030.
    public static let deltaT: Double = 69

    public static func from(_ date: Date) -> Double { date.timeIntervalSince1970 / 86400 + 2440587.5 }

    public static func date(_ jd: Double) -> Date { Date(timeIntervalSince1970: (jd - 2440587.5) * 86400) }

    /// Gregorian calendar date with fractional day (Meeus 7.1).
    public static func calendar(year: Int, month: Int, day: Double) -> Double {
        var y = Double(year), m = Double(month)
        if month <= 2 { y -= 1; m += 12 }
        let a = (y / 100).rounded(.down)
        let b = 2 - a + (a / 4).rounded(.down)
        return (365.25 * (y + 4716)).rounded(.down) + (30.6001 * (m + 1)).rounded(.down) + day + b - 1524.5
    }

    /// Julian Ephemeris Day for a UT instant.
    public static func ephemeris(_ date: Date) -> Double { from(date) + deltaT / 86400 }

    /// Julian centuries since J2000.0.
    public static func centuries(fromJDE jde: Double) -> Double { (jde - 2451545) / 36525 }
}
```

```swift
import Foundation

/// Mean sidereal time in degrees (Meeus 12.4).
public enum SiderealTime {
    public static func greenwichMean(_ date: Date) -> Double {
        let jd = JulianDay.from(date)
        let t = (jd - 2451545) / 36525
        return normalizedDegrees(280.46061837 + 360.98564736629 * (jd - 2451545)
                                 + 0.000387933 * t * t - t * t * t / 38710000)
    }

    /// Longitude east-positive.
    public static func localMean(_ date: Date, longitude: Double) -> Double {
        normalizedDegrees(greenwichMean(date) + longitude)
    }
}
```

`Coordinates.swift`:

```swift
import Foundation

/// Right ascension and declination in degrees.
public struct Equatorial: Equatable, Sendable {
    public var ra: Double
    public var dec: Double
    public init(ra: Double, dec: Double) { self.ra = ra; self.dec = dec }
}

/// Altitude above the horizon and azimuth from north through east, in degrees.
public struct Horizontal: Equatable, Sendable {
    public var altitude: Double
    public var azimuth: Double
    public init(altitude: Double, azimuth: Double) { self.altitude = altitude; self.azimuth = azimuth }
}

/// Observing site; longitude east-positive. Privacy: never logged (ADR 0006), so the
/// descriptions are redacted.
public struct Site: Codable, Equatable, Sendable, CustomStringConvertible, CustomDebugStringConvertible {
    public let latitude: Double
    public let longitude: Double
    public let elevation: Double

    public init(latitude: Double, longitude: Double, elevation: Double) {
        self.latitude = latitude
        self.longitude = longitude
        self.elevation = elevation
    }

    /// The site as N.I.N.A. reports it; `0, 0, 0` (no site set in N.I.N.A.) and missing
    /// fields are no site.
    public static func reported(latitude: Double?, longitude: Double?, elevation: Double?) -> Site? {
        guard let latitude, let longitude else { return nil }
        if latitude == 0, longitude == 0, (elevation ?? 0) == 0 { return nil }
        return Site(latitude: latitude, longitude: longitude, elevation: elevation ?? 0)
    }

    public var description: String { "Site(redacted)" }
    public var debugDescription: String { "Site(redacted)" }
}

public enum Sky {
    /// Meeus 13.5, 13.6 with azimuth from north.
    public static func horizontal(_ eq: Equatorial, latitude: Double, localSiderealTime: Double) -> Horizontal {
        let h = localSiderealTime - eq.ra
        let altitude = asinDeg(sinDeg(latitude) * sinDeg(eq.dec) + cosDeg(latitude) * cosDeg(eq.dec) * cosDeg(h))
        let fromSouth = atan2Deg(sinDeg(h), cosDeg(h) * sinDeg(latitude) - tanDeg(eq.dec) * cosDeg(latitude))
        return Horizontal(altitude: altitude, azimuth: normalizedDegrees(fromSouth + 180))
    }

    public static func horizontal(_ eq: Equatorial, site: Site, at date: Date) -> Horizontal {
        horizontal(eq, latitude: site.latitude, localSiderealTime: SiderealTime.localMean(date, longitude: site.longitude))
    }

    /// Hour angle in −180 ..< 180; 0 at the upper meridian transit.
    public static func hourAngle(_ eq: Equatorial, site: Site, at date: Date) -> Double {
        let h = normalizedDegrees(SiderealTime.localMean(date, longitude: site.longitude) - eq.ra)
        return h >= 180 ? h - 360 : h
    }

    /// Rigorous precession from J2000.0 to the mean equinox of the date (Meeus 21.2–21.4).
    public static func precessFromJ2000(_ eq: Equatorial, to date: Date) -> Equatorial {
        let t = JulianDay.centuries(fromJDE: JulianDay.ephemeris(date))
        let zeta = (2306.2181 * t + 0.30188 * t * t + 0.017998 * t * t * t) / 3600
        let z = (2306.2181 * t + 1.09468 * t * t + 0.018203 * t * t * t) / 3600
        let theta = (2004.3109 * t - 0.42665 * t * t - 0.041833 * t * t * t) / 3600
        let a = cosDeg(eq.dec) * sinDeg(eq.ra + zeta)
        let b = cosDeg(theta) * cosDeg(eq.dec) * cosDeg(eq.ra + zeta) - sinDeg(theta) * sinDeg(eq.dec)
        let c = sinDeg(theta) * cosDeg(eq.dec) * cosDeg(eq.ra + zeta) + cosDeg(theta) * sinDeg(eq.dec)
        return Equatorial(ra: normalizedDegrees(atan2Deg(a, b) + z), dec: asinDeg(c))
    }

    /// Angular distance (Meeus 17.1).
    public static func separation(_ a: Equatorial, _ b: Equatorial) -> Double {
        let d = sinDeg(a.dec) * sinDeg(b.dec) + cosDeg(a.dec) * cosDeg(b.dec) * cosDeg(a.ra - b.ra)
        return acos(max(-1, min(1, d))) * 180 / .pi
    }

    /// Nutation in longitude and obliquity in degrees, accurate to 0.5″ (Meeus ch. 22).
    public static func nutation(jde: Double) -> (longitude: Double, obliquity: Double) {
        let t = JulianDay.centuries(fromJDE: jde)
        let omega = 125.04452 - 1934.136261 * t
        let l = 280.4665 + 36000.7698 * t
        let lm = 218.3165 + 481267.8813 * t
        let dpsi = -17.20 * sinDeg(omega) - 1.32 * sinDeg(2 * l) - 0.23 * sinDeg(2 * lm) + 0.21 * sinDeg(2 * omega)
        let deps = 9.20 * cosDeg(omega) + 0.57 * cosDeg(2 * l) + 0.10 * cosDeg(2 * lm) - 0.09 * cosDeg(2 * omega)
        return (dpsi / 3600, deps / 3600)
    }

    /// Mean obliquity of the ecliptic (Meeus 22.2).
    public static func meanObliquity(jde: Double) -> Double {
        let t = JulianDay.centuries(fromJDE: jde)
        return 23.43929111 - (46.8150 * t + 0.00059 * t * t - 0.001813 * t * t * t) / 3600
    }

    /// Ecliptic → equatorial (Meeus 13.3, 13.4).
    public static func equatorial(eclipticLongitude l: Double, latitude b: Double, obliquity e: Double) -> Equatorial {
        let ra = atan2Deg(sinDeg(l) * cosDeg(e) - tanDeg(b) * sinDeg(e), cosDeg(l))
        let dec = asinDeg(sinDeg(b) * cosDeg(e) + cosDeg(b) * sinDeg(e) * sinDeg(l))
        return Equatorial(ra: normalizedDegrees(ra), dec: dec)
    }
}
```

```swift
import Foundation

/// Apparent position of the Sun, low accuracy (≈ 0.01°; Meeus ch. 25).
public enum Sun {
    public static func position(jde: Double) -> (equatorial: Equatorial, longitude: Double) {
        let t = JulianDay.centuries(fromJDE: jde)
        let l0 = 280.46646 + 36000.76983 * t + 0.0003032 * t * t
        let m = 357.52911 + 35999.05029 * t - 0.0001537 * t * t
        let c = (1.914602 - 0.004817 * t - 0.000014 * t * t) * sinDeg(m)
            + (0.019993 - 0.000101 * t) * sinDeg(2 * m) + 0.000289 * sinDeg(3 * m)
        let omega = 125.04 - 1934.136 * t
        let lambda = l0 + c - 0.00569 - 0.00478 * sinDeg(omega)
        let epsilon = Sky.meanObliquity(jde: jde) + 0.00256 * cosDeg(omega)
        let ra = atan2Deg(cosDeg(epsilon) * sinDeg(lambda), cosDeg(lambda))
        let dec = asinDeg(sinDeg(epsilon) * sinDeg(lambda))
        return (Equatorial(ra: normalizedDegrees(ra), dec: dec), normalizedDegrees(lambda))
    }

    public static func position(at date: Date) -> Equatorial { position(jde: JulianDay.ephemeris(date)).equatorial }

    /// Geometric altitude of the Sun's centre (parallax 0.0024° neglected).
    public static func horizontal(site: Site, at date: Date) -> Horizontal {
        Sky.horizontal(position(at: date), site: site, at: date)
    }
}
```

```swift
            sl += l * a.factor(m) * sinDeg(arg)
            sr += r * a.factor(m) * cosDeg(arg)

        return (Sky.equatorial(eclipticLongitude: lambda, latitude: geo.latitude, obliquity: epsilon),

    }

        let moon = position(at: date)
        let geo = Sky.horizontal(moon.equatorial, site: site, at: date)
        let parallax = asinDeg(6378.14 / moon.distanceKm)
        return Horizontal(altitude: geo.altitude - parallax * cosDeg(geo.altitude), azimuth: geo.azimuth)
    }

        let i = 180 - a.d - 6.289 * sinDeg(a.mp) + 2.100 * sinDeg(a.m) - 1.274 * sinDeg(2 * a.d - a.mp)

    static func elongation(at date: Date) -> Double {
        let jde = JulianDay.ephemeris(date)
        return normalizedDegrees(ecliptic(jde: jde).longitude - Sun.position(jde: jde).longitude)
    }

    public static func isWaxing(at date: Date) -> Bool { elongation(at: date) < 180 }

    /// Next instant the elongation reaches 180°, to about a minute.
    public static func nextFullMoon(after date: Date) -> Date {
        var start = date
        var previous = elongation(at: start)
        for _ in 0 ..< 31 * 4 {
            let end = start.addingTimeInterval(6 * 3600)
            let next = elongation(at: end)
            if previous < 180, next >= 180 || next < previous {
                var low = start, high = end
                while high.timeIntervalSince(low) > 30 {
                    let mid = low.addingTimeInterval(high.timeIntervalSince(low) / 2)
                    let e = elongation(at: mid)
                    if e < 180, e >= elongation(at: low) { low = mid } else { high = mid }
                }
                return low
            }
            start = end
            previous = next
        }
        return start
    }
}
```

```swift
import Foundation

public struct Crossing: Equatable, Sendable {
    public let date: Date
    public let rising: Bool
}

/// Instants when an altitude crosses a threshold: sampled every `step`, then bisected to 1 s.
public enum Crossings {
    public static func find(in interval: DateInterval, threshold: Double, step: TimeInterval = 600,
                            altitude: (Date) -> Double) -> [Crossing] {
        var result: [Crossing] = []
        var t0 = interval.start
        var above0 = altitude(t0) > threshold
        while t0 < interval.end {
            let t1 = min(t0.addingTimeInterval(step), interval.end)
            let above1 = altitude(t1) > threshold
            if above1 != above0 {
                var low = t0, high = t1
                while high.timeIntervalSince(low) > 1 {
                    let mid = low.addingTimeInterval(high.timeIntervalSince(low) / 2)
                    if (altitude(mid) > threshold) == above0 { low = mid } else { high = mid }
                }
                result.append(Crossing(date: high, rising: above1))
            }
            t0 = t1
            above0 = above1
        }
        return result
    }
}
```

```swift
import Foundation

/// Sun and Moon events of one night, from local noon to the next local noon.
public struct NightEvents: Equatable, Sendable {
    public static let horizonAltitude = -0.833
    public static let civilAltitude = -6.0
    public static let nauticalAltitude = -12.0
    public static let astronomicalAltitude = -18.0
    public static let moonAltitude = 0.125

    public let window: DateInterval
    public let sunset: Date?
    public let sunrise: Date?
    public let civilDusk: Date?
    public let civilDawn: Date?
    public let nauticalDusk: Date?
    public let nauticalDawn: Date?
    public let astronomicalDusk: Date?
    public let astronomicalDawn: Date?
    public let moonrise: Date?
    public let moonset: Date?
    public let sunAlwaysUp: Bool
    public let sunAlwaysDown: Bool

    /// The night that contains `date`: from local noon (today, or yesterday before noon)
    /// to the next local noon, DST-safe.
    public static func window(containing date: Date, timeZone: TimeZone) -> DateInterval {
        var calendar = Calendar(identifier: .gregorian)
        calendar.timeZone = timeZone
        let noon = calendar.date(bySettingHour: 12, minute: 0, second: 0, of: date)!
        let start = date >= noon ? noon : calendar.date(byAdding: .day, value: -1, to: noon)!
        let end = calendar.date(byAdding: .day, value: 1, to: start)!
        return DateInterval(start: start, end: end)
    }

    public static func compute(site: Site, containing date: Date, timeZone: TimeZone) -> NightEvents {
        let window = window(containing: date, timeZone: timeZone)
        let sunAltitude = { (d: Date) in Sun.horizontal(site: site, at: d).altitude }
        func duskDawn(_ threshold: Double) -> (Date?, Date?) {
            let crossings = Crossings.find(in: window, threshold: threshold, altitude: sunAltitude)
            let dusk = crossings.first { !$0.rising }
            let dawn = crossings.first { $0.rising && (dusk == nil || $0.date > dusk!.date) }
            return (dusk?.date, dawn?.date)
        }
        let horizon = duskDawn(horizonAltitude)
        let civil = duskDawn(civilAltitude)
        let nautical = duskDawn(nauticalAltitude)
        let astronomical = duskDawn(astronomicalAltitude)
        let moon = Crossings.find(in: window, threshold: moonAltitude) { Moon.geocentricAltitude(site: site, at: $0) }
        let noHorizonCrossing = horizon.0 == nil && horizon.1 == nil
        let startsUp = sunAltitude(window.start) > horizonAltitude
        return NightEvents(window: window, sunset: horizon.0, sunrise: horizon.1,
                           civilDusk: civil.0, civilDawn: civil.1,
                           nauticalDusk: nautical.0, nauticalDawn: nautical.1,
                           astronomicalDusk: astronomical.0, astronomicalDawn: astronomical.1,
                           moonrise: moon.first { $0.rising }?.date, moonset: moon.first { !$0.rising }?.date,
                           sunAlwaysUp: noHorizonCrossing && startsUp, sunAlwaysDown: noHorizonCrossing && !startsUp)
    }
}
```

```swift
import Foundation

public struct AltitudePoint: Equatable, Sendable {
    public let date: Date
    public let altitude: Double
}

/// Altitudes every 5 minutes, both ends included; geometric (no refraction).
public enum AltitudeCurve {
    public static let step: TimeInterval = 300

    static func sample(_ interval: DateInterval, _ altitude: (Date) -> Double) -> [AltitudePoint] {
        let count = Int((interval.duration / step).rounded(.down))
        var points = (0 ... count).map { i -> AltitudePoint in
            let date = interval.start.addingTimeInterval(Double(i) * step)
            return AltitudePoint(date: date, altitude: altitude(date))
        }
        if let last = points.last, last.date < interval.end {
            points.append(AltitudePoint(date: interval.end, altitude: altitude(interval.end)))
        }
        return points
    }

    /// `ofDate`: coordinates of the equinox of date (precess J2000 targets once per night).
    public static func target(_ ofDate: Equatorial, site: Site, over interval: DateInterval) -> [AltitudePoint] {
        sample(interval) { Sky.horizontal(ofDate, site: site, at: $0).altitude }
    }

    public static func sun(site: Site, over interval: DateInterval) -> [AltitudePoint] {
        sample(interval) { Sun.horizontal(site: site, at: $0).altitude }
    }

    /// Topocentric.
    public static func moon(site: Site, over interval: DateInterval) -> [AltitudePoint] {
        sample(interval) { Moon.horizontal(site: site, at: $0).altitude }
    }

    /// Upper meridian transit in the interval: the hour angle passes from negative to positive.
    public static func transit(_ ofDate: Equatorial, site: Site, over interval: DateInterval) -> Date? {
        let crossing = Crossings.find(in: interval, threshold: 0, step: 600) { date in
            let h = Sky.hourAngle(ofDate, site: site, at: date)
            // Fold the lower transit (±180°) away so only the upper one is a crossing.
            return abs(h) > 90 ? -1 : h
        }
        return crossing.first { $0.rising }?.date
    }
}
```

```swift
import Foundation

/// What the altitude curve follows: the sequence's target container (J2000, from
/// `/sequence/state` `Target.InputCoordinates`), else the tracking mount (JNOW).
public struct NightTarget: Equatable, Sendable {
    public enum Source: Equatable, Sendable { case sequence, mount }
    public enum Epoch: Equatable, Sendable { case j2000, ofDate }

    public let name: String?
    public let coordinates: Equatorial
    public let epoch: Epoch
    public let source: Source

    public init(name: String?, coordinates: Equatorial, epoch: Epoch, source: Source) {
        self.name = name
        self.coordinates = coordinates
        self.epoch = epoch
        self.source = source
    }

    /// Coordinates for the equinox of `date`.
    public func ofDate(_ date: Date) -> Equatorial {
        epoch == .j2000 ? Sky.precessFromJ2000(coordinates, to: date) : coordinates
    }

    public static func fromSequence(state: JSONValue) -> NightTarget? {
        var running: NightTarget?
        var children = state.array ?? []
        while let node = children.first(where: { $0["Status"]?.string == "RUNNING" && $0["Items"]?.array != nil }) {
            if let target = target(of: node) { running = target }
            children = node["Items"]?.array ?? []
        }
        if let running { return running }
        return firstCreated(in: state.array ?? [])
    }

    private static func firstCreated(in nodes: [JSONValue]) -> NightTarget? {
        for node in nodes {
            if node["Status"]?.string == "CREATED", let target = target(of: node) { return target }
            if let items = node["Items"]?.array, let found = firstCreated(in: items) { return found }
        }
        return nil
    }

    private static func target(of container: JSONValue) -> NightTarget? {
        guard let input = container["Target"]?["InputCoordinates"],
              let rh = input["RAHours"]?.double, let rm = input["RAMinutes"]?.double, let rs = input["RASeconds"]?.double,
              let dd = input["DecDegrees"]?.double, let dm = input["DecMinutes"]?.double, let ds = input["DecSeconds"]?.double
        else { return nil }
        let negative = input["NegativeDec"]?.bool == true || dd < 0
        let dec = (abs(dd) + dm / 60 + ds / 3600) * (negative ? -1 : 1)
        let containerName = container["Name"]?.string.map { $0.hasSuffix("_Container") ? String($0.dropLast(10)) : $0 }
        let name = [container["Target"]?["TargetName"]?.string, containerName].compactMap { $0 }.first { !$0.isEmpty }
        return NightTarget(name: name, coordinates: Equatorial(ra: (rh + rm / 60 + rs / 3600) * 15, dec: dec),
                           epoch: .j2000, source: .sequence)
    }

    public static func fromMount(_ mount: NINAEquipmentSummary.Mount?, name: String?) -> NightTarget? {
        guard let mount, mount.activity == .tracking,
              let ra = mount.rightAscensionHours, let dec = mount.declinationDegrees else { return nil }
        return NightTarget(name: name, coordinates: Equatorial(ra: ra * 15, dec: dec), epoch: .ofDate, source: .mount)
    }

    public static func resolve(sequence: NightTarget?, mount: NINAEquipmentSummary.Mount?, imageTarget: String?) -> NightTarget? {
        sequence ?? fromMount(mount, name: imageTarget)
    }
}
```

```swift
    case cached(reported: Date)

/// The last site a connected mount reported, kept on the device (ADR 0006).
public struct CachedSite: Codable, Equatable, Sendable {
    public let site: Site
    public let reported: Date
    public init(site: Site, reported: Date) { self.site = site; self.reported = reported }
}

public enum SiteResolution {
    /// The mount always wins; the cache covers the hours the mount is off.
    public static func resolve(mount: Site?, cached: CachedSite?) -> (site: Site, source: SiteSource)? {
        if let mount { return (mount, .mount) }
        if let cached { return (cached.site, .cached(reported: cached.reported)) }
        return nil
    }

/// Everything about one night that does not change by the minute; computed once per
/// night, target or site.
public struct NightPlan: Equatable, Sendable {
    public enum TargetVisibility: Equatable, Sendable { case visible, neverRises }
    public enum Darkness: Equatable, Sendable {
        case astronomical(from: Date, until: Date)
        case nautical(from: Date, until: Date)
        case none
    }

    public let site: Site
    public let target: NightTarget?
    public let events: NightEvents
    public let chartRange: DateInterval
    public let sunCurve: [AltitudePoint]
    public let moonCurve: [AltitudePoint]
    public let targetCurve: [AltitudePoint]?
    public let transit: Date?
    public let targetVisibility: TargetVisibility?
    /// Target coordinates of date (middle of the night); `nil` without target.
    let targetOfDate: Equatorial?

    public var darkness: Darkness {
        if let from = events.astronomicalDusk, let until = events.astronomicalDawn { return .astronomical(from: from, until: until) }
        if let from = events.nauticalDusk, let until = events.nauticalDawn { return .nautical(from: from, until: until) }
        return .none
    }

    public static func compute(site: Site, target: NightTarget?, containing date: Date, timeZone: TimeZone) -> NightPlan {
        let events = NightEvents.compute(site: site, containing: date, timeZone: timeZone)
        let window = events.window
        let start = max(window.start, events.sunset?.addingTimeInterval(-7200) ?? window.start)
        let end = min(window.end, events.sunrise?.addingTimeInterval(3600) ?? window.end)
        let range = start < end ? DateInterval(start: start, end: end) : window
        let ofDate = target?.ofDate(window.start.addingTimeInterval(window.duration / 2))
        let visibility: TargetVisibility? = ofDate.map { eq in
            AltitudeCurve.target(eq, site: site, over: window).allSatisfy { $0.altitude <= 0 } ? .neverRises : .visible
        }
        return NightPlan(site: site, target: target, events: events, chartRange: range,
                         sunCurve: AltitudeCurve.sun(site: site, over: range),
                         moonCurve: AltitudeCurve.moon(site: site, over: range),
                         targetCurve: ofDate.map { AltitudeCurve.target($0, site: site, over: range) },
                         transit: ofDate.flatMap { AltitudeCurve.transit($0, site: site, over: window) },
                         targetVisibility: visibility, targetOfDate: ofDate)
    }

    public func isCurrent(site: Site, target: NightTarget?, at date: Date, timeZone: TimeZone) -> Bool {
        self.site == site && self.target == target && events.window.contains(date)
            && events.window == NightEvents.window(containing: date, timeZone: timeZone)
    }
}

/// Values that move with the clock; recomputed once a minute.
public struct NightNow: Equatable, Sendable {
    public let sun: Horizontal
    public let moon: Horizontal
    public let moonIlluminated: Double
    public let moonWaxing: Bool
    public let nextFullMoon: Date
    public let target: Horizontal?
    public let moonSeparation: Double?
    public let nextMoonEvent: Crossing?

    public static func compute(plan: NightPlan, at date: Date) -> NightNow {
        let site = plan.site
        let moonPosition = Moon.position(at: date)
        let events = [plan.events.moonrise.map { Crossing(date: $0, rising: true) },
                      plan.events.moonset.map { Crossing(date: $0, rising: false) }]
            .compactMap { $0 }.filter { $0.date > date }.sorted { $0.date < $1.date }
        return NightNow(sun: Sun.horizontal(site: site, at: date),
                        moon: Moon.horizontal(site: site, at: date),
                        moonIlluminated: Moon.illuminatedFraction(at: date),
                        moonWaxing: Moon.isWaxing(at: date),
                        nextFullMoon: Moon.nextFullMoon(after: date),
                        target: plan.targetOfDate.map { Sky.horizontal($0, site: site, at: date) },
                        moonSeparation: plan.targetOfDate.map { Sky.separation(moonPosition.equatorial, $0) },
                        nextMoonEvent: events.first)
    }
}
```

```swift
import Foundation
import PanGalacticCore

/// Where cached sites live; one dictionary per device (ADR 0006).
public protocol SiteStorage: Sendable {
    func load() -> [UUID: CachedSite]
    func save(_ sites: [UUID: CachedSite])
}

public struct UserDefaultsSiteStorage: SiteStorage, @unchecked Sendable {
    public static let key = "site.v1"
    private let defaults: UserDefaults

    public init(defaults: UserDefaults = .standard) { self.defaults = defaults }

    public func load() -> [UUID: CachedSite] {
        guard let data = defaults.data(forKey: Self.key),
              let decoded = try? JSONDecoder().decode([String: CachedSite].self, from: data) else { return [:] }
        return Dictionary(uniqueKeysWithValues: decoded.compactMap { key, value in UUID(uuidString: key).map { ($0, value) } })
    }

    public func save(_ sites: [UUID: CachedSite]) {
        let encodable = Dictionary(uniqueKeysWithValues: sites.map { ($0.key.uuidString, $0.value) })
        if let data = try? JSONEncoder().encode(encodable) { defaults.set(data, forKey: Self.key) }
    }
}

public final class InMemorySiteStorage: SiteStorage, @unchecked Sendable {
    private let lock = NSLock()
    private var stored: [UUID: CachedSite] = [:]
    public init() {}
    public func load() -> [UUID: CachedSite] { lock.withLock { stored } }
    public func save(_ sites: [UUID: CachedSite]) { lock.withLock { stored = sites } }
}

/// The last site each instance's mount reported, so the night page works while the
/// mount is off. Never logged; never leaves the device.
@MainActor
public final class SiteStore {
    private let storage: any SiteStorage
    private var sites: [UUID: CachedSite]

    public init(storage: any SiteStorage = UserDefaultsSiteStorage()) {
        self.storage = storage
        sites = storage.load()
    }

    public func site(for id: UUID) -> CachedSite? { sites[id] }

    public func record(_ site: Site, for id: UUID, at date: Date) {
        if let old = sites[id], old.site == site, date.timeIntervalSince(old.reported) < 3600 { return }
        sites[id] = CachedSite(site: site, reported: date)
        storage.save(sites)
    }

    public func remove(id: UUID) {
        guard sites.removeValue(forKey: id) != nil else { return }
        storage.save(sites)
    }
}
```

```markdown
# ADR 0006 – The site is cached on the device

- Status: Accepted (owner decision)
- Date: 2026-11-20

## Context

M4.1 computes the night (Sun, Moon, darkness, the target's altitude) from the site. N.I.N.A. reports the site only through `mount/info` (`SiteLatitude`, `SiteLongitude`, `SiteElevation`) and only while the mount is connected; by day or with the mount off the app would not know it. The site is privacy-sensitive (AGENTS.md); the owner runs several observatories and already decided that coordinates may be shown on the TV (capability matrix §7, 2026-11-16).

## Decision

- The last site a connected mount reported is kept per instance on the Apple TV in `UserDefaults` under `site.v1` (`[instance UUID: {site, reported}]`), next to the instances (ADR 0003).
- The mount always wins; the cached site is used only while the mount reports none, and the night page says "Site last reported on <date>".
- `0, 0, 0` (no site set in N.I.N.A.) is never stored.
- Removing an instance removes its site.
- The site is never logged, exported, put in fixtures, screenshots or issues, or sent anywhere; `Site`'s descriptions are redacted.
- No manual entry (the owner chose the cache over typing coordinates with the remote).

## Consequences

- The night page works by day and with the mount off once the mount was connected one time.
- A moved telescope shows the old site until its mount reports the new one; the date on the page makes that visible.
```

```swift
                             at: CGPoint(x: c.x + (r - 22) * sin(a), y: c.y - (r - 22) * cos(a)))
            }
            func ring(_ h: Horizontal, _ color: Color) {
                let p = point(h)
                let rect = CGRect(x: p.x - 10, y: p.y - 10, width: 20, height: 20)
                context.stroke(Path(ellipseIn: rect), with: .color(color),
                               style: StrokeStyle(lineWidth: 3, dash: h.altitude < 0 ? [4, 3] : []))
            }
            ring(now.sun, Theme.warn)
            ring(now.moon, Theme.text2)
            if let telescope {
                let p = point(telescope)
                context.fill(Path(ellipseIn: CGRect(x: p.x - 12, y: p.y - 12, width: 24, height: 24)), with: .color(Theme.accent))
            }
        }
        .accessibilityHidden(true)
    }

    /// Eight-point compass name for an azimuth (N, NE, E, … in the UI language).
    static func point(_ azimuth: Double) -> LocalizedStringKey {
        let names: [LocalizedStringKey] = ["N", "NE", "E", "SE", "S", "SW", "W", "NW"]
        return names[Int(((azimuth + 22.5).truncatingRemainder(dividingBy: 360)) / 45)]
    }
}
```
